HUNGARIAN(G):
1  h ← DEFAULT-FEASIBLE-LABELING(G)
2  M ← GREEDY-BIPARTITE-MATCHING(Gh)
3  form Gh and GM,h
4  while M is not perfect:
5    P ← FIND-AUGMENTING-PATH(GM,h)
6    M ← M ⊕ P
7  return M
